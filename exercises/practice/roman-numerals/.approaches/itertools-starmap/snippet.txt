def roman(number):
    orders = [(1000, 'M  '), (100, 'CDM'), (10, 'XLC'), (1, 'IVX')]
    def options(i, v, x):
        return ['', i, i * 2, i * 3, i + v, v, v + i, v + i * 2, v + i * 3, i + x]
    def compute(val, chars):
        return options(*chars)[number % (val * 10) // val]
    return ''.join(starmap(compute, orders))
